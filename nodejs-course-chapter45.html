Authentication
با JWT

یکی از پرتکرارترین موضوعات مصاحبه Node.js — هش کردن پسورد با bcrypt، و ساخت سیستم Authentication با Access/Refresh Token.

bcryptJWTAccess TokenRefresh Token

Authentication چیست؟

Authentication یعنی تشخیص اینکه کاربر واقعاً کیست — «شما کی هستید؟». این با Authorization (فصل بعد) فرق دارد که می‌پرسد «شما اجازه چه کاری را دارید؟». امروز رایج‌ترین روش Authentication در API‌ها، JWT (JSON Web Token) است.

هش کردن پسورد — هرگز پسورد خام ذخیره نکنید

اگر دیتابیس شما هک شود و پسوردها به صورت متن ساده ذخیره شده باشند، فاجعه است. باید پسورد را hash کنید — یک تابع یک‌طرفه که برگشت‌ناپذیر است.

terminal
npm install bcrypt # یا جایگزین مدرن‌تر و امن‌تر: npm install argon2
services/passwordService.js
const bcrypt = require('bcrypt'); // SALT_ROUNDS — هرچقدر بیشتر، امن‌تر ولی کندتر. 10-12 معمول است const SALT_ROUNDS = 12; async function hashPassword(plainPassword) { // bcrypt خودکار یک "salt" تصادفی هم اضافه می‌کند // یعنی حتی دو پسورد یکسان، hash متفاوت تولید می‌کنند return bcrypt.hash(plainPassword, SALT_ROUNDS); } async function verifyPassword(plainPassword, hashedPassword) { // مقایسه — هیچ‌وقت hash را decode نمی‌کنید، فقط مقایسه می‌کنید return bcrypt.compare(plainPassword, hashedPassword); } // استفاده در ثبت‌نام async function registerUser(email, plainPassword) { const hashedPassword = await hashPassword(plainPassword); return User.create({ email, password: hashedPassword }); // در دیتابیس فقط hash ذخیره می‌شود — چیزی شبیه: // $2b$12$KIXxPfnK3P8mHIrPTMTU4uJhh4TQdrBz7... } // استفاده در لاگین async function loginUser(email, plainPassword) { const user = await User.findOne({ email }).select('+password'); if (!user) throw new Error('ایمیل یا رمز اشتباه است'); const isValid = await verifyPassword(plainPassword, user.password); if (!isValid) throw new Error('ایمیل یا رمز اشتباه است'); // پیام یکسان برای هر دو خطا — تا مهاجم نفهمد ایمیل درست است یا نه return user; }
⚠️

هیچ‌وقت از MD5 یا SHA256 خام برای پسورد استفاده نکنید — این‌ها سریع هستند و برای crack شدن با brute-force مناسب‌اند. bcrypt/argon2 عمداً کند طراحی شده‌اند.

JWT — JSON Web Token

JWT یک token خودمختار (self-contained) است — همه اطلاعات لازم برای شناسایی کاربر داخل خودش دارد، و سرور نیازی به نگه‌داری session در حافظه ندارد (Stateless — یادتان هست از فصل 12 Factor؟).

ساختار JWT: سه بخش با نقطه جدا شده: header.payload.signature. Header و payload فقط base64 encode شده‌اند (نه رمزنگاری!) — هرکسی می‌تواند بخواندشان. امنیت از signature می‌آید که با یک secret فقط سرور امضا می‌کند.

terminal
npm install jsonwebtoken
services/authService.js
const jwt = require('jsonwebtoken'); // ساخت Access Token — کوتاه‌مدت (15 دقیقه تا 1 ساعت) function generateAccessToken(user) { return jwt.sign( { userId: user.id, email: user.email, role: user.role }, // payload process.env.JWT_SECRET, // secret — باید طولانی و تصادفی باشد { expiresIn: '15m' } // کوتاه‌مدت — امنیت بیشتر ); } // ساخت Refresh Token — طولانی‌مدت (روزها تا هفته‌ها) // وقتی Access Token منقضی شد، با این یکی، Access Token جدید می‌گیریم function generateRefreshToken(user) { return jwt.sign( { userId: user.id }, process.env.JWT_REFRESH_SECRET, // secret متفاوت از access token { expiresIn: '7d' } ); } // verify — چک کردن اعتبار token function verifyAccessToken(token) { try { return jwt.verify(token, process.env.JWT_SECRET); // اگر امضا نامعتبر یا منقضی باشد، throw می‌کند } catch (error) { throw new Error('Token نامعتبر یا منقضی شده'); } }

چرا دو Token؟ (Access + Refresh)

Access Token کوتاه‌مدت یعنی اگر لو برود، سریع منقضی می‌شود — آسیب محدود است. Refresh Token طولانی‌مدت کاربر را مجبور به لاگین دوباره هر ۱۵ دقیقه نمی‌کند — فقط وقتی access token expired شد، با refresh token یک access token تازه می‌گیرید.

routes/auth.js — چرخه کامل
// POST /auth/login async function login(req, res) { const user = await loginUser(req.body.email, req.body.password); const accessToken = generateAccessToken(user); const refreshToken = generateRefreshToken(user); // refresh token را در httpOnly cookie بگذارید — جاوااسکریپت client نمی‌تواند بخواندش // این از حملات XSS محافظت می‌کند res.cookie('refreshToken', refreshToken, { httpOnly: true, secure: true, // فقط HTTPS sameSite: 'strict', // محافظت در برابر CSRF maxAge: 7 * 24 * 60 * 60 * 1000 }); res.json({ accessToken, user: { id: user.id, email: user.email } }); } // POST /auth/refresh — گرفتن access token جدید async function refresh(req, res) { const refreshToken = req.cookies.refreshToken; if (!refreshToken) return res.status(401).json({ error: 'Refresh token یافت نشد' }); try { const payload = jwt.verify(refreshToken, process.env.JWT_REFRESH_SECRET); const user = await User.findById(payload.userId); const newAccessToken = generateAccessToken(user); res.json({ accessToken: newAccessToken }); } catch { res.status(401).json({ error: 'Refresh token نامعتبر است' }); } }

Middleware احراز هویت

middleware/authenticate.js
function authenticate(req, res, next) { const authHeader = req.headers.authorization; // "Bearer eyJhbGc..." const token = authHeader?.split(' ')[1]; if (!token) { return res.status(401).json({ error: 'Token لازم است' }); } try { req.user = verifyAccessToken(token); // اطلاعات کاربر روی request قرار می‌گیرد next(); } catch (error) { res.status(401).json({ error: error.message }); } } // استفاده app.get('/profile', authenticate, (req, res) => { res.json({ userId: req.user.userId }); // از middleware می‌آید });

نکات کلیدی برای مصاحبه

  • Authentication = «شما کی هستید؟». Authorization = «چه اجازه‌ای دارید؟» (فصل بعد).
  • پسورد را همیشه با bcrypt/argon2 هش کنید — هرگز MD5/SHA خام، هرگز متن ساده.
  • JWT سه بخش دارد: header.payload.signature — payload رمزنگاری نشده، فقط امضا شده.
  • Access Token کوتاه‌مدت + Refresh Token طولانی‌مدت — تعادل بین امنیت و تجربه کاربری.
  • Refresh token را در httpOnly cookie نگه دارید — نه در localStorage (خطر XSS).
  • پیام خطای لاگین باید یکسان باشد برای ایمیل غلط و پسورد غلط — جلوگیری از user enumeration.
  • سوال رایج: «چرا JWT payload را می‌شود خواند ولی نمی‌شود تغییر داد؟» — payload فقط با base64 کد شده (نه رمزنگاری)، پس هرکسی می‌تواند بخواندش؛ ولی هر تغییری در payload باعث می‌شود امضا (signature) دیگر با محتوا مطابقت نداشته باشد و سرور آن را رد کند.
  • یک تله رایج: ذخیره اطلاعات حساس (مثل پسورد یا شماره کارت) داخل payload JWT — چون قابل خواندن است، هرگز نباید داده حساس آنجا باشد.
  • bcrypt عمداً کند طراحی شده (با salt rounds قابل تنظیم) تا حمله brute-force را کند و پرهزینه کند — این ویژگی اصلی است که آن را از هش‌های ساده مثل SHA256 متمایز می‌کند.
  • چرا دو نوع Token؟ اگر Access Token لو برود، به خاطر کوتاه بودن عمرش آسیب محدود است؛ اگر همیشه یک Token طولانی‌مدت استفاده کنید، لو رفتنش خطر خیلی بیشتری دارد.
